# Глобальные объекты
Все нижеописанные объекты находятся в глобальном пространстве любой сцены.

```lua
return {
    "display", --создание объектов
    "sound", --создание звуков
    "window", --управление окном
    "physics", --управление физикой -
    "render", --управление рендером -
    "events", --создание/подключение и управление событиями
    "mouse", --управление мышью
    "fileTool", --модуль управления файлами
    "overray", --прослойка над графическим движком что даёт возможность рисовать в обход -

    --== Глобальные функции ==--
    "update -->" = 0.016, --вызывается каждый кадр, 0.016 - время между кадрами.
    "mouseMoved -->" = {
        x = 0, --позиция мыши X
        y = 0, --позиция мыши Y
        dx = 0, --дельта мыши X
        dy = 0, --дельта мыши Y
    },
    "onMouseWheel -->" = 0, --вызывается при действии колёсика мыши, возвращает соответствующее значение.
    "mouseClick -->" = { --вызывается при клике мышью
        button = "left/right/midle",
        phase = "pressed/down/released"
    },
    "charPressed -->" = "char", --вызывается при нажатии клавиши, возвращает её реальное назначение.
    "keyboard -->" = { --вызывается при нажатии клавиши, возвращает её индекс.
        phase = "down/pressed/released",
        key = "key"
    },
    "tween(GameObject,'key',number duration,number endValue,easeType,listener())" --функция используемая для создания плавных анимаций, подробнее в tween.
}
```
